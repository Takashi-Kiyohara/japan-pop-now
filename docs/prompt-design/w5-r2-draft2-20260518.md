## 削除 sub-routing (R1 §3 + R2 Y-2 修正、Draft 3 P-3 で 3-stage 化)

```

  endDate < today - 60d AND category in [collab-cafe, popup, limited-event]
  → 410 Gone (canonical 汚染回避、48h で deindex、stage B より優先)

stage B (replacement、stage A FALSE のみ):
  frontmatter.supersededBy != null AND topic equivalent
  → 301 (link equity 移譲)
  ★ 同一性 manual gate: script は `proposed_301_target` 提示、user approve 必須

stage C (default、stage A∧B FALSE):
  → noindex frontmatter + sitemap 除外 + 90 日 TTL
  ★ 90 日 grace 期間中に書き直し未実施なら stage A 410 自動昇格
  ★ orphan noindex の HCU site-quality re-degradation 防止
```

## 7 軸 score 関数 spec (実装の核)

### scoreA (実訪問)
```js
function scoreA(article, slug) {
  const heroBodyImages = extractImages(article);
  const firstPartyCount = heroBodyImages.filter(isFirstParty).length;
  const firstPartyRatio = firstPartyCount / heroBodyImages.length;
  const firstHandPara = countFirstHandParagraphs(article.content);
  return {
    pass: firstPartyRatio >= 0.4 && firstHandPara >= 1,
    firstPartyRatio, firstHandPara,
  };
}

function isFirstParty(imgSrc) {
  if (imgSrc.match(/wikimedia\.org|commons\.wikimedia/)) return false;
  if (imgSrc.match(/unsplash|pexels|shutterstock/)) return false;
  if (imgSrc.match(/^\/images\/articles\/[^/]+\/(IMG_|moe-shot|takashi-)/)) return true;
  return null;  // 判定不能、auto-FAIL に倒す
}
```

### scoreB (プレス言換、R2 R-1 修正)
```js
async function scoreB(article, slug, pressMap, competitorCache) {
  const emb = await getEmbedder();
  const cleanContent = stripSections(article.content).slice(0, 2000);

  // 1. press release fetch (primary)
  const pressUrl = pressMap[slug];
  if (pressUrl) {
    try {
      const pressText = await fetchPressBody(pressUrl);
      if (pressText) {
        const v1 = await emb(cleanContent, { pooling: 'mean', normalize: true });
        const v2 = await emb(pressText.slice(0, 2000), { pooling: 'mean', normalize: true });
        const cosine = dot(v1.data, v2.data);
        return { pass: cosine < 0.72, cosine, source: 'press' };
      }
    } catch (e) {}
  }

  // 2. competitor top10 fallback
  try {
    const competitorTexts = await fetchCompetitorTop10(article.frontmatter.title, competitorCache);
    if (competitorTexts.length > 0) {
      let maxCosine = 0;
      const v1 = await emb(cleanContent, { pooling: 'mean', normalize: true });
      for (const t of competitorTexts) {
        const v2 = await emb(t.slice(0, 2000), { pooling: 'mean', normalize: true });
        maxCosine = Math.max(maxCosine, dot(v1.data, v2.data));
      }
      return { pass: maxCosine < 0.55, cosine: maxCosine, source: 'competitor_top10' };
    }
  } catch (e) {}

  // 3. P-1: fallback の fallback = manual_SME_flag
  return { pass: null, cosine: null, source: 'manual_SME_flag', reason: 'both press and competitor fetch failed' };
}
```

### scoreE (AI fp、R2 R-1 + Draft 3 P-4)
```js
async function scoreE(article, slug, baseline) {
  const body = stripSections(article.content);
  const emDashDensity = countEmDash(body) / wordCount(body) * 100;
  const bpHits = countBoilerplate(body, NICHE_WHITELIST);
  const sd = stdDevSentenceLength(body);

  const emDashZ = (emDashDensity - baseline.em_dash_density.mu) / baseline.em_dash_density.sigma;
  const bpZ = (bpHits - baseline.boilerplate_hits.mu) / baseline.boilerplate_hits.sigma;
  const sdZ = (sd - baseline.sentence_length_variance.mu) / baseline.sentence_length_variance.sigma;

  const compositePass = emDashZ < 2 && bpZ < 2 && sdZ > -2;

  // P-4: suspect 15% に圧縮、GPTZero 2nd opinion
  if (!compositePass && process.env.GPTZERO_API_KEY) {
    const isSuspect = (emDashZ >= 2 && emDashZ < 3) || (bpZ >= 2 && bpZ < 3) || (sdZ <= -2 && sdZ > -3);
    if (isSuspect) {
      const gptz = await fetchGPTZero(body.slice(0, 3000));
      if (gptz?.probability_ai < 0.5) return { pass: true, source: 'gptzero_2nd' };
    }
  }

  return { pass: compositePass, em_dash_density: emDashDensity, bp_hits: bpHits, sd, emDashZ, bpZ, sdZ };
}
```

### scoreF (Helpfulness、R1 §1)
```js
function scoreF(article, slug) {
  const required = {
    tldr: /^##\s+TL;DR/im.test(article.content) || /^##\s+概要/im.test(article.content),
    price: /^##\s+価格/im.test(article.content) || /^##\s+Price/im.test(article.content) || /￥|¥|\$/.test(article.content),
    hours: /営業時間|営業日|Hours|Opening hours/i.test(article.content),
    booking: /予約|reservation|booking|klook|kkday/i.test(article.content) || /(klook|kkday)\.com/.test(article.content),
    access: /アクセス|Access|address|nearest station|徒歩.*分/i.test(article.content),
  };
  const count = Object.values(required).filter(Boolean).length;
  return { pass: count >= 4, required, count };
}
```

### scoreG (Author binding、R1 §1)
```js
function scoreG(article, slug) {
  const firstHandPara = countFirstHandParagraphs(article.content);
  return { pass: firstHandPara >= 1, firstHandPara };
}

function countFirstHandParagraphs(content) {
  const para = content.split(/\n\n+/);
  return para.filter(p =>
    /(私が|私は|I visited|I went|I tried|I checked|on (Jan|Feb|Mar|Apr|May|Jun|Jul|Aug|Sep|Oct|Nov|Dec) \d+)/.test(p)
  ).length;
}
```

## decideBucket (R2 R-4)
```js
function decideBucket(scores, passCount, frontmatter, today) {
  const a = scores.A.pass, b = scores.B.pass, c = scores.C.pass, g = scores.G.pass;
  const f = scores.F.pass;

  if (passCount <= 2 || (a === false && b === false && c === false && g === false)) {
    return deleteRouting(frontmatter, today);
  }
  if (passCount >= 6 && a && f && g) {
    return { type: 'maintain' };
  }
  return { type: 'fix', noindex_quarantine: true, manual_reaudit_gate: true };
}

function deleteRouting(frontmatter, today) {
  const endDate = frontmatter.endDate ? new Date(frontmatter.endDate) : null;
  const todayDate = new Date(today);

  // stage A
  const eventBound = endDate &&
    (todayDate - endDate) > 60 * 86400000 &&
    ['collab-cafe', 'popup', 'limited-event'].includes(frontmatter.category);
  if (eventBound) {
    return { type: 'delete', stage: 'A', routing: '410', reason: 'event ended 60d+ ago' };
  }

  // stage B
  if (frontmatter.supersededBy) {
    return {
      type: 'delete', stage: 'B', routing: '301',
      target: frontmatter.supersededBy,
      manual_gate: true,
    };
  }

  // stage C (default)
  return {
    type: 'delete', stage: 'C', routing: 'noindex',
    ttl_expires: new Date(todayDate.getTime() + 90 * 86400000).toISOString(),
    auto_promote: 'stage_A_410',
  };
}
```

## main() 構造
```js
async function main() {
  const today = process.env.AUDIT_TODAY || new Date().toISOString();
  const competitorCache = process.env.COMPETITOR_CACHE || '/tmp/w5-competitor-cache.json';
  const baseline = await calibrate();
  const articles = await loadAllArticles();

  const results = [];
  for (const article of articles) {
    const slug = article.frontmatter.slug;
    const scores = {
      A: await scoreA(article, slug),
      B: await scoreB(article, slug, pressMap, competitorCache),
      C: scoreC(article, slug),
      D: await scoreD(article, slug),
      E: await scoreE(article, slug, baseline),
      F: scoreF(article, slug),
      G: scoreG(article, slug),
    };
    const passCount = Object.values(scores).filter(s => s.pass === true).length;
    const bucket = decideBucket(scores, passCount, article.frontmatter, today);
    results.push({ slug, scores, passCount, bucket });
  }

  await writeResultJson(results);
  await writeResultMarkdown(results);
}
```

## w5-calibrate.mjs (5 firsthand article から μ+σ 抽出)
```js
import calibration from '../../lib/audit/calibration-baseline.json' assert { type: 'json' };

async function calibrate() {
  const stats = { em_dash_density: [], boilerplate_hits: [], sentence_length_variance: [] };
  for (const slug of calibration.human_confirmed_articles) {
    const md = await fs.readFile(`content/articles/${slug}.mdx`, 'utf-8');
    const { content } = matter(md);
    const body = stripSections(content);
    stats.em_dash_density.push(countEmDash(body) / wordCount(body) * 100);
    stats.boilerplate_hits.push(countBoilerplate(body, NICHE_WHITELIST));
    stats.sentence_length_variance.push(stdDevSentenceLength(body));
  }
  return {
    em_dash_density: { mu: mean(stats.em_dash_density), sigma: stdDev(stats.em_dash_density) },
    boilerplate_hits: { mu: mean(stats.boilerplate_hits), sigma: stdDev(stats.boilerplate_hits) },
    sentence_length_variance: { mu: mean(stats.sentence_length_variance), sigma: stdDev(stats.sentence_length_variance) },
  };
}
```
