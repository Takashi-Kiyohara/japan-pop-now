#### 1-1 scoreB patch (P-1: scraping fail → manual_SME_flag)
```js
async function scoreB(article, slug) {
  const pressUrl = PRESS_MAP[slug];
  const emb = await getEmbedder();
  const cleanContent = stripSections(article.content).slice(0, 2000);

  // 1. press release fetch (primary)
  if (pressUrl) {
    try {
      const pressText = await fetchPressBody(pressUrl);
      if (pressText) {
        const v1 = await emb(cleanContent, { pooling: 'mean', normalize: true });
        const v2 = await emb(pressText.slice(0, 2000), { pooling: 'mean', normalize: true });
        const cosine = dot(v1.data, v2.data);
        return { pass: cosine < 0.72, cosine, source: 'press' };
      }
    } catch (e) {
      // press fetch fail → fallback to competitor
    }
  }

  // 2. competitor top10 fallback
  try {
    const competitorTexts = await fetchCompetitorTop10(article.frontmatter.title);
    if (competitorTexts.length > 0) {
      let maxCosine = 0;
      const v1 = await emb(cleanContent, { pooling: 'mean', normalize: true });
      for (const t of competitorTexts) {
        const v2 = await emb(t.slice(0, 2000), { pooling: 'mean', normalize: true });
        maxCosine = Math.max(maxCosine, dot(v1.data, v2.data));
      }
      return { pass: maxCosine < 0.55, cosine: maxCosine, source: 'competitor_top10' };
    }
  } catch (e) {
    // competitor scraping fail → manual_SME_flag
  }

  // 3. P-1: fallback の fallback = manual_SME_flag
  return { pass: null, cosine: null, source: 'manual_SME_flag', reason: 'both press and competitor fetch failed, manual review required' };
}
```

#### 1-2 scoreE patch (P-4: suspect 15%)
```js
async function scoreE(article, slug) {
  // ... (Draft 2 と同じ 3-feature composite + calibration)
  const compositePass = emDashZ < 2 && bpZ < 2 && sdZ > -2;

  // P-4: dual-pass 範囲を 30% → 15% に圧縮 (free tier 内化)
  if (!compositePass && process.env.GPTZERO_API_KEY) {
    // suspect zone: composite 失敗のうち、Z score が 2-3σ 範囲 (= 上位 15% suspect、上位 30% より絞り込み)
    const isSuspect = (emDashZ >= 2 && emDashZ < 3) || (bpZ >= 2 && bpZ < 3) || (sdZ <= -2 && sdZ > -3);
    if (isSuspect) {
      const gptz = await fetchGPTZero(stripSections(article.content).slice(0, 3000));
      if (gptz?.probability_ai < 0.5) return { pass: true, source: 'gptzero_2nd' };
    }
  }

  return { pass: compositePass, em_dash_density: emDashDensity, bp_hits: bpHits, sd: stddev };
}
```

function deleteRouting(article, frontmatter) {
  const endDate = frontmatter.endDate ? new Date(frontmatter.endDate) : null;
  const today = new Date();

  // stage B 優先 (event-bound + supersededBy の場合): link equity > canonical 汚染
  if (frontmatter.supersededBy) {
    return {
      routing: '301',
      target: frontmatter.supersededBy,
      manual_gate: true,  // user approve 必須
      auto_shortlist_eligible: checkSameIPSameVenueWithin90d(article, frontmatter.supersededBy),
    };
  }

  // stage A (event-bound 強制 410)
  const eventBound = endDate &&
    (today - endDate) > 60 * 86400000 &&
    ['collab-cafe', 'popup', 'limited-event'].includes(frontmatter.category);
  if (eventBound) {
    return { routing: '410', reason: 'event ended 60d+ ago' };
  }

  // stage C (default): noindex with 90日 TTL
  return {
    routing: 'noindex',
    ttl_expires: new Date(today.getTime() + 90 * 86400000),

}
